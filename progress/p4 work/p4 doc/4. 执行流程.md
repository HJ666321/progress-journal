# 无条件执行
- apply(forward_dst):很明显这个我不用管
``` p4
// 定义一个名为 ac_set_egr 的动作，它接受一个名为 egress_spec 的参数。
action ac_set_egr(egress_spec) {
    // 调用内置函数 modify_field 来修改字段。
    // 第一个参数是要修改的字段：这是Tofino交换机内部的、用于单播流量管理的标准元数据字段，全称为 "ingress intrinsic metadata for traffic manager's unicast egress port"，即指定数据包的出端口。
    // 第二个参数是 egress_spec，即上面传进来的目标端口号。
    modify_field(ig_intr_md_for_tm.ucast_egress_port, egress_spec);
}

// 定义一个名为 forward_dst 的表。
table forward_dst {
    // "reads" 块定义了查表的"键"（key）。
    reads {
        // 使用以太网头部的目的MAC地址（eth.dst）作为精确匹配（exact）的键。
        eth.dst : exact;
    }
    // "actions" 块定义了这张表可以执行哪些动作。
    actions {
        // 可以执行我们上面定义的 ac_set_egr 动作。
        ac_set_egr; 
        // 也可以执行一个什么都不做的空动作 ac_nop（这个动作在main.p4或其他文件中定义）。
        ac_nop;
    }
    // 如果查表没有匹配到任何条目，则执行默认动作。
    default_action : ac_nop;
    // 设置这张表可以容纳的条目数量为64。
    size : 64;
}
```
